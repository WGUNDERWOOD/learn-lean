\begin{frame}{The Mathlib repository}

  \begin{itemize}
    \item Mathlib is a library (or package) for Lean, started in 2017.
    \item Serves as a repository of formalised mathematics.
    \item Now contains 200\,000 theorems and 2.4 million lines of code.
    \item Includes measure theory and probability but not statistics.
    \item E.g., the strong law of large numbers is
      \texttt{strong\_law\_ae}.
    \item Mathlib also provides tooling to make Lean easier to use.
  \end{itemize}

\begin{frame}{Writing our first proofs}

  \vspace*{2mm}

  \begin{itemize}
    \item \texttt{example}:
      Introduce a new proposition to prove.
    \item \texttt{rfl}:
      Prove a proposition by definition.
    \item \texttt{exact h}:
      Prove a proposition by applying hypothesis \texttt{h}.
    \item \texttt{rw [h]}:
      Rewrite the goal using hypothesis \texttt{h}.
    \item Comments can be written with: \texttt{-- here is a comment}.
    \item \texttt{\#check}:
      Check the type of an expression. \\
      \texttt{\#check 1} returns \texttt{1:}$\N$. \\
      \texttt{\#check (1 + 1 = 2)} returns \texttt{(1 + 1 = 2):Prop}.
    \item \texttt{constructor}:
      Transform \texttt{P and Q} into two goals, \texttt{P} and \texttt{Q}.
    \item \texttt{sorry}:
      Temporarily prove any goal (and raise a warning).
    \item \texttt{have}:
      Introduce a new sub-goal to prove.
  \end{itemize}

\end{frame}

\begin{frame}{Basics of type-theoretic logic}

  \begin{itemize}
    \item The Curry--Howard correspondence gives an isomorphism between
      models for \emph{type theory} and \emph{logic}.
  \end{itemize}

  \begin{table}[H]
    \begin{center}
      \begin{tabular}{|c|c|}
        \hline
        \textbf{Type theory} & \textbf{Logic} \\
        \hline
        Type \texttt{T} & Proposition \texttt{P} \\
        Element of type \texttt{x:T} & Proof of proposition \texttt{h:P} \\
        \texttt{n:}$\N$ & \texttt{rfl:(1 + 1 = 2)} \\
        \hline
      \end{tabular}
    \end{center}
  \end{table}

  \begin{itemize}
    \item The proposition \texttt{True} has a proof
      \texttt{trivial:True}.
    \item The proposition \texttt{False} has no proofs!
  \end{itemize}

\end{frame}

\begin{frame}{Basics of type-theoretic logic}

  \begin{itemize}
    \item If \texttt{P} and \texttt{Q} are propositions,
      then a proof of \texttt{(P}$\implies$\texttt{Q)} \\
      is a function mapping proofs of \texttt{P}
      to proofs of \texttt{Q}.
    \item A proof of \texttt{(P and Q)} is a pair of proofs: \\
      \texttt{(hP, hQ):(P and Q)} where
      \texttt{hP:P} and \texttt{hQ:Q}.
    \item A proof of \texttt{(not P)} maps
      proofs of \texttt{P} to proofs of \texttt{False}.
    \item \texttt{(P or Q)} is more complicated:
      we may prove \texttt{(P or Q)} without a proof either
      for \texttt{P} or for \texttt{Q}. For example:
      %
      \begin{align*}
        \exists p, q \in \R \setminus \Q:
        p^q \in \Q.
      \end{align*}
      \emph{Proof.} \
      Consider $x \coloneqq \sqrt{2}^{\sqrt{2}}$ and
      $x^{\sqrt 2}$.
      \qed
  \end{itemize}
\end{frame}

\begin{frame}[plain]{A more interesting hand-written proof}

  %We present a famous result and proof due to Euclid,
  %published in the Elements, around 300 BCE.

  \vspace*{0mm}

  \begin{beamertheorem}[Infinitude of the primes]
    For all $n \in \N$,
    there exists $p \in \N$ such that
    $p > n$ and $p$ is prime.
  \end{beamertheorem}

  \vspace*{-1mm}

  \begin{proof}
    %
    Take $n \in \N$, write $m \coloneqq n! + 1$,
    and let $p$ be a prime factor of $m$.
    \vspace*{-3mm}
    %
    \begin{itemize}
        \setlength{\itemsep}{0mm}
      \item Note that $m \neq 1$ because $n! \neq 0$.
      \item Therefore, $p$ is prime.
      \item Suppose for contradiction that we do not have $p > n$.
      \item Then $p \leq n$.
      \item Note that $p > 0$ because $p$ is a factor of $m$.
      \item We have $p \mid m$ by definition of $p$.
      \item Also, $p \mid n!$ because $p \leq n$ by assumption and $p > 0$.
      \item So $p \mid 1$ by modular arithmetic.
      \item Thus, $p = 1$.
      \item But $1$ is not prime \textreferencemark.
        {\addtobeamertemplate{qed symbol}{}{\hspace*{2mm}}\qedhere}
    \end{itemize}
    %
  \end{proof}

\end{frame}

\begin{frame}{Writing an AI-assisted proof}

  \begin{beamertheorem}
    Let $X_1, \ldots, X_n$ be independent and
    identically distributed random variables
    with mean $\E(X_1) = 0$ and variance
    $\E(X_1^2) = \sigma^2 < \infty$.
    Then for all $\varepsilon > 0$, we have
    %
    \begin{equation*}
      \mathbb{P}\Biggl(
        \biggl|
        \frac{1}{n}
        \sum_{i=1}^n
        X_i
        \biggr|
        > \varepsilon
      \Biggr)
      \leq
      \frac{\sigma^2}{n \varepsilon^2}.
    \end{equation*}
    %
  \end{beamertheorem}

\begin{frame}{Getting help}

  \begin{itemize}
    \item \texttt{\#check}:
      Check the type of an expression. \\
      Propositions and proofs have types
      and can be checked! \\
      For example, \texttt{\#check Nat.eq\_one\_of\_dvd\_one}
    \item \texttt{\#eval}:
      Evaluate some Lean code. \\
      For example, \texttt{\#eval Nat.minFac 1}
    \item Advanced tactics for finding proofs: \\
      \texttt{aesop}, \texttt{omega}, \texttt{linarith},
      \texttt{ring}, \texttt{grind}
    \item \url{https://leansearch.net/}: \\
      Search Mathlib using natural language.
    \item \url{github.com/leanprover-community/mathlib4} \\
      The Mathlib code repository.
    \item \url{https://lean-lang.org/}:
      Core documentation for Lean.
  \end{itemize}
